<script setup lang="ts">
import { computed, onMounted, ref, watch } from "vue";

import BaseEmptyState from "@/components/base/BaseEmptyState.vue";
import BasePageHeader from "@/components/base/BasePageHeader.vue";
import BasePagination from "@/components/base/BasePagination.vue";

import ChannelCard from "@/components/iptv/ChannelCard.vue";
import ChannelDetailDialog from "@/components/iptv/ChannelDetailDialog.vue";
import ChannelEditDialog from "@/components/iptv/ChannelEditDialog.vue";
import ChannelFilters from "@/components/iptv/ChannelFilters.vue";
import GridSizeSlider from "@/components/iptv/GridSizeSlider.vue";
import M3uImportDialog from "@/components/iptv/M3uImportDialog.vue";

import {
  iptvChannelRepository,
  countryRepository,
  languageRepository,
  iptvSourceRepository,
  tagRepository,
  iptvCategoryRepository,
} from "@/domain/repositories";

import {
  usedCategories,
  usedCountries,
  usedQualities,
} from "@/domain/iptv/filterOptions";

import type {
  IptvCategoryRecord,
  IptvChannelRecord,
  IptvSourceRecord,
  NamedRecord,
} from "@/types/records";

type IptvViewMode = "grid" | "logos" | "list" | "compact";

const IPTV_GRID_WIDTH_STORAGE_KEY = "mediario-iptv-grid-width";
const MIN_GRID_CARD_WIDTH = 170;
const MAX_GRID_CARD_WIDTH = 420;
const DEFAULT_GRID_CARD_WIDTH = 250;
const CHANNELS_PER_PAGE = 18;

const channels = ref<IptvChannelRecord[]>([]);
const allChannels = ref<IptvChannelRecord[]>([]);

const page = ref(1);
const pages = ref(1);
const total = ref(0);

const loading = ref(true);
const error = ref("");

const search = ref("");
const country = ref("");
const category = ref("");
const quality = ref("");
const status = ref("");
const favorite = ref("");

const view = ref<IptvViewMode>("grid");

const storedGridCardWidth = Number(
  localStorage.getItem(IPTV_GRID_WIDTH_STORAGE_KEY),
);

const gridCardWidth = ref<number>(
  Math.min(
    MAX_GRID_CARD_WIDTH,
    Math.max(
      MIN_GRID_CARD_WIDTH,
      storedGridCardWidth || DEFAULT_GRID_CARD_WIDTH,
    ),
  ),
);

const countries = ref<NamedRecord[]>([]);
const languages = ref<NamedRecord[]>([]);
const tags = ref<NamedRecord[]>([]);
const categories = ref<IptvCategoryRecord[]>([]);

const selectedChannel = ref<IptvChannelRecord | null>(null);
const editingChannel = ref<IptvChannelRecord | null>(null);

const channelSources = ref<IptvSourceRecord[]>([]);
const sourcesLoading = ref(false);
const importDialogOpen = ref(false);

let filterTimer: ReturnType<typeof setTimeout> | undefined;

const countryOptions = computed(() =>
  usedCountries(allChannels.value, countries.value),
);

const categoryOptions = computed(() =>
  usedCategories(allChannels.value, categories.value),
);

const qualityOptions = computed(() =>
  usedQualities(allChannels.value),
);

const filterCount = computed(() =>
  [
    search.value,
    country.value,
    category.value,
    quality.value,
    status.value,
    favorite.value,
  ].filter(Boolean).length,
);

const channelCountDescription = computed(() =>
  total.value === 1
    ? "1 canal configurado"
    : `${total.value} canais configurados`,
);

function escapeFilter(value: string): string {
  return value
    .replaceAll("\\", "\\\\")
    .replaceAll('"', '\\"');
}

function buildChannelFilter(): string {
  const terms = ["cemiterio = false"];
  const searchTerm = search.value.trim();

  if (searchTerm) {
    const escapedSearch = escapeFilter(searchTerm);

    terms.push(
      `(nome ~ "${escapedSearch}" || tvg_name ~ "${escapedSearch}" || tvg_id ~ "${escapedSearch}" || epg_id ~ "${escapedSearch}" || playlist_origem ~ "${escapedSearch}")`,
    );
  }

  if (country.value) {
    terms.push(`pais = "${escapeFilter(country.value)}"`);
  }

  if (category.value) {
    terms.push(`categoria = "${escapeFilter(category.value)}"`);
  }

  if (quality.value) {
    terms.push(`qualidade = "${escapeFilter(quality.value)}"`);
  }

  if (status.value) {
    terms.push(`ativo = ${status.value === "active"}`);
  }

  if (favorite.value) {
    terms.push(`favorito = ${favorite.value === "yes"}`);
  }

  return terms.join(" && ");
}

async function loadLibrary(): Promise<void> {
  loading.value = true;
  error.value = "";

  try {
    const result = await iptvChannelRepository.list({
      page: page.value,
      perPage: CHANNELS_PER_PAGE,
      filter: buildChannelFilter(),
      sort: "ordem,nome",
      expand: "pais,categoria,idiomas,tags",
    });

    channels.value = result.items;
    pages.value = result.totalPages || 1;
    total.value = result.totalItems;
  } catch (cause) {
    error.value =
      cause instanceof Error
        ? cause.message
        : String(cause);

    channels.value = [];
    pages.value = 1;
    total.value = 0;
  } finally {
    loading.value = false;
  }
}

async function loadAllChannels(): Promise<void> {
  const result: IptvChannelRecord[] = [];

  let currentPage = 1;
  let totalPages = 1;

  do {
    const response = await iptvChannelRepository.list({
      page: currentPage,
      perPage: 500,
      filter: "cemiterio = false",
      sort: "nome",
    });

    result.push(...response.items);

    totalPages = response.totalPages || 1;
    currentPage += 1;
  } while (currentPage <= totalPages);

  allChannels.value = result;
}

async function loadLookups(): Promise<void> {
  const [
    countryResult,
    categoryResult,
    languageResult,
    tagResult,
  ] = await Promise.all([
    countryRepository.list({
      page: 1,
      perPage: 250,
      filter: "ativo = true",
      sort: "ordem,nome",
    }),
    iptvCategoryRepository.list({
      page: 1,
      perPage: 100,
      filter: "ativo = true",
      sort: "ordem,nome",
    }),
    languageRepository.list({
      page: 1,
      perPage: 100,
      filter: "ativo = true",
      sort: "ordem,nome",
    }),
    tagRepository.list({
      page: 1,
      perPage: 200,
      filter: "ativo = true",
      sort: "ordem,nome",
    }),
  ]);

  countries.value = countryResult.items;
  categories.value = categoryResult.items;
  languages.value = languageResult.items;
  tags.value = tagResult.items;
}

async function openChannel(
  channel: IptvChannelRecord,
): Promise<void> {
  selectedChannel.value = channel;
  channelSources.value = [];
  sourcesLoading.value = true;

  try {
    const result = await iptvSourceRepository.list({
      page: 1,
      perPage: 100,
      filter: `canal_tv = "${channel.id}"`,
      sort: "prioridade",
    });

    channelSources.value = result.items;
  } catch (cause) {
    error.value =
      cause instanceof Error
        ? cause.message
        : String(cause);

    channelSources.value = [];
  } finally {
    sourcesLoading.value = false;
  }
}

function updateChannel(
  channel: IptvChannelRecord,
): void {
  channels.value = channels.value.map((currentChannel) =>
    currentChannel.id === channel.id
      ? channel
      : currentChannel,
  );

  allChannels.value = allChannels.value.map((currentChannel) =>
    currentChannel.id === channel.id
      ? channel
      : currentChannel,
  );

  if (selectedChannel.value?.id === channel.id) {
    selectedChannel.value = channel;
  }
}

function handleChannelSaved(
  channel: IptvChannelRecord,
): void {
  updateChannel(channel);
  editingChannel.value = null;
}

function handleChannelArchived(id: string): void {
  channels.value = channels.value.filter(
    (channel) => channel.id !== id,
  );

  allChannels.value = allChannels.value.filter(
    (channel) => channel.id !== id,
  );

  if (selectedChannel.value?.id === id) {
    selectedChannel.value = null;
    channelSources.value = [];
  }

  editingChannel.value = null;
  total.value = Math.max(0, total.value - 1);
}


async function handleM3uImported(count: number): Promise<void> {
  importDialogOpen.value = false;

  if (count <= 0) return;

  page.value = 1;
  await Promise.all([
    loadAllChannels(),
    loadLibrary(),
  ]);
}
function clearFilters(): void {
  search.value = "";
  country.value = "";
  category.value = "";
  quality.value = "";
  status.value = "";
  favorite.value = "";

  page.value = 1;

  void loadLibrary();
}

function changePage(nextPage: number): void {
  page.value = nextPage;
  void loadLibrary();
}

watch(gridCardWidth, (value) => {
  localStorage.setItem(
    IPTV_GRID_WIDTH_STORAGE_KEY,
    String(value),
  );
});

watch(
  [
    search,
    country,
    category,
    quality,
    status,
    favorite,
  ],
  () => {
    if (filterTimer) {
      clearTimeout(filterTimer);
    }

    filterTimer = setTimeout(() => {
      page.value = 1;
      void loadLibrary();
    }, 250);
  },
);

watch(countryOptions, (options) => {
  if (
    country.value &&
    !options.some((option) => option.value === country.value)
  ) {
    country.value = "";
  }
});

watch(categoryOptions, (options) => {
  if (
    category.value &&
    !options.some((option) => option.value === category.value)
  ) {
    category.value = "";
  }
});

watch(qualityOptions, (options) => {
  if (
    quality.value &&
    !options.some((option) => option.value === quality.value)
  ) {
    quality.value = "";
  }
});

onMounted(async () => {
  try {
    await Promise.all([
      loadLookups(),
      loadAllChannels(),
    ]);

    await loadLibrary();
  } catch (cause) {
    error.value =
      cause instanceof Error
        ? cause.message
        : String(cause);

    loading.value = false;
  }
});
</script>

/**
 * Ficheiro: src/domain/iptv/filterOptions.ts
 * Tipo: Transformação TypeScript pura.
 * Função: Gera filtros exclusivamente com valores realmente usados.
 */
import type { ChannelRecord, NamedRecord, TvCategoryRecord } from "@/types/records";
export interface UsedFilterOption { value:string; label:string; count:number }
function count(channels:ChannelRecord[],field:"pais"|"categoria"){const map=new Map<string,number>();for(const item of channels){const id=item[field];if(id)map.set(id,(map.get(id)||0)+1)}return map}
function clean(items:UsedFilterOption[]){return items.filter(x=>x.value&&x.label.trim()&&x.count>0).sort((a,b)=>a.label.localeCompare(b.label,"pt-PT"))}
export function usedCountries(channels:ChannelRecord[],lookups:NamedRecord[]){const map=count(channels,"pais");return clean(lookups.map(x=>({value:x.id,label:x.nome||x.codigo||"",count:map.get(x.id)||0})))}
export function usedCategories(channels:ChannelRecord[],lookups:TvCategoryRecord[]){const map=count(channels,"categoria");return clean(lookups.map(x=>({value:x.id,label:x.nome||"",count:map.get(x.id)||0})))}
export function usedQualities(channels:ChannelRecord[]){const map=new Map<string,number>();for(const item of channels){const value=item.qualidade||"Desconhecida";map.set(value,(map.get(value)||0)+1)}return clean([...map].map(([value,count])=>({value,label:value==="FHD"?"Full HD":value,count})))}

<script setup lang="ts">
import { computed, nextTick, ref, watch } from "vue";

import {
  iptvCategoryRepository,
  iptvChannelRepository,
  iptvSourceRepository,
} from "@/domain/repositories";
import { parseM3u, type M3uItem } from "@/domain/iptv/m3u";
import { findM3uDuplicate } from "@/domain/iptv/m3uDuplicate";

import type {
  IptvCategoryRecord,
  IptvChannelRecord,
} from "@/types/records";

interface ImportFailure {
  name: string;
  message: string;
}

interface ImportSummary {
  imported: number;
  skipped: number;
  failed: number;
  failures: ImportFailure[];
  categoriesCreated: number;
  internalDuplicates: number;
}

const props = defineProps<{
  open: boolean;
  existing: IptvChannelRecord[];
}>();

const emit = defineEmits<{
  close: [];
  imported: [count: number];
}>();

const firstInput = ref<HTMLInputElement | null>(null);
const items = ref<M3uItem[]>([]);
const fileName = ref("");
const error = ref("");
const importing = ref(false);
const progress = ref(0);
const batchTotal = ref(0);
const query = ref("");
const summary = ref<ImportSummary | null>(null);

const visible = computed(() => {
  const normalizedQuery = query.value.toLocaleLowerCase("pt-PT");

  return items.value.filter((item) => {
    if (!normalizedQuery) return true;

    return [item.name, item.groupTitle, item.tvgId].some((value) =>
      value.toLocaleLowerCase("pt-PT").includes(normalizedQuery),
    );
  });
});

const selected = computed(() =>
  items.value.filter((item) => item.selected && !item.errors.length),
);

const invalidCount = computed(() =>
  items.value.filter((item) => item.errors.length).length,
);

function reset(): void {
  items.value = [];
  fileName.value = "";
  error.value = "";
  importing.value = false;
  progress.value = 0;
  batchTotal.value = 0;
  query.value = "";
  summary.value = null;
}

watch(
  () => props.open,
  async (open) => {
    if (!open) return;

    reset();
    await nextTick();
    firstInput.value?.focus();
  },
);

async function readFile(event: Event): Promise<void> {
  const input = event.target as HTMLInputElement;
  const file = input.files?.[0];

  if (!file) return;

  fileName.value = file.name;
  error.value = "";
  progress.value = 0;
  batchTotal.value = 0;
  summary.value = null;

  try {
    items.value = parseM3u(await file.text());
  } catch (cause) {
    items.value = [];
    error.value = cause instanceof Error ? cause.message : String(cause);
  }
}

function selectAll(value: boolean): void {
  for (const item of visible.value) {
    item.selected = value && !item.errors.length;
  }
}

function sourceType(url: string): "hls" | "http" {
  return /\.m3u8(?:$|[?#])/i.test(url) ? "hls" : "http";
}

function normalizedKey(value: string): string {
  return value
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLocaleLowerCase("pt-PT")
    .replace(/[^a-z0-9]+/g, " ")
    .trim();
}

function internalDuplicateKeys(entries: M3uItem[]): Set<M3uItem> {
  const seenUrls = new Set<string>();
  const seenTvgIds = new Set<string>();
  const duplicates = new Set<M3uItem>();

  for (const item of entries) {
    const url = item.url.trim();
    const tvgId = normalizedKey(item.tvgId);

    if (seenUrls.has(url) || (tvgId && seenTvgIds.has(tvgId))) {
      duplicates.add(item);
      continue;
    }

    seenUrls.add(url);
    if (tvgId) seenTvgIds.add(tvgId);
  }

  return duplicates;
}

async function loadCategoryMap(): Promise<Map<string, IptvCategoryRecord>> {
  const result = await iptvCategoryRepository.list({
    page: 1,
    perPage: 500,
    sort: "ordem,nome",
  });

  return new Map(
    result.items.map((category) => [normalizedKey(category.nome), category]),
  );
}

async function categoryFor(
  groupTitle: string,
  categories: Map<string, IptvCategoryRecord>,
): Promise<{ id?: string; created: boolean }> {
  const name = groupTitle.trim();
  if (!name) return { created: false };

  const key = normalizedKey(name);
  const existing = categories.get(key);
  if (existing) return { id: existing.id, created: false };

  const created = await iptvCategoryRepository.create({
    nome: name,
    ativo: true,
    ordem: 0,
  });
  categories.set(key, created);
  return { id: created.id, created: true };
}

async function run(): Promise<void> {
  const entries = [...selected.value];

  if (!entries.length) return;

  importing.value = true;
  progress.value = 0;
  batchTotal.value = entries.length;
  error.value = "";
  summary.value = null;

  let importedCount = 0;
  let skippedCount = 0;
  let categoriesCreated = 0;
  const failures: ImportFailure[] = [];
  const duplicateItems = internalDuplicateKeys(entries);
  const categories = await loadCategoryMap();

  for (const item of entries) {
    if (duplicateItems.has(item)) {
      skippedCount += 1;
      progress.value += 1;
      continue;
    }
    const duplicate = findM3uDuplicate(item, props.existing);

    if (duplicate.kind && duplicate.kind !== "name") {
      skippedCount += 1;
      progress.value += 1;
      continue;
    }

    let channel: IptvChannelRecord | null = null;

    try {
      const category = await categoryFor(item.groupTitle, categories);
      if (category.created) categoriesCreated += 1;

      channel = (await iptvChannelRepository.create({
        nome: item.name,
        url_stream: item.url,
        tvg_id: item.tvgId,
        tvg_name: item.tvgName,
        tvg_logo_url: item.tvgLogo,
        playlist_origem: fileName.value,
        adulto: item.adult,
        ativo: true,
        cemiterio: false,
        categoria: category.id,
        qualidade: "Desconhecida",
      })) as IptvChannelRecord;

      await iptvSourceRepository.create({
        canal_tv: channel.id,
        url: item.url,
        rotulo: "Importado da playlist",
        tipo: sourceType(item.url),
        qualidade: "Desconhecida",
        prioridade: 0,
        ativo: true,
      });

      importedCount += 1;
    } catch (cause) {
      let message = cause instanceof Error ? cause.message : String(cause);

      if (channel?.id) {
        try {
          await iptvChannelRepository.delete(channel.id);
        } catch (rollbackCause) {
          const rollbackMessage =
            rollbackCause instanceof Error
              ? rollbackCause.message
              : String(rollbackCause);

          message +=
            ` O canal criado não pôde ser removido: ${rollbackMessage}`;
        }
      }

      failures.push({
        name: item.name,
        message,
      });
    } finally {
      progress.value += 1;
    }
  }

  summary.value = {
    imported: importedCount,
    skipped: skippedCount,
    failed: failures.length,
    failures,
    categoriesCreated,
    internalDuplicates: duplicateItems.size,
  };

  importing.value = false;

  if (importedCount > 0) {
    emit("imported", importedCount);
  }
}
</script>

import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import {

import { expectNoA11yViolations } from "@/test/accessibility";
import M3uImportDialog from "./M3uImportDialog.vue";

const mocks = vi.hoisted(() => ({
  channelCreate: vi.fn(),
  channelDelete: vi.fn(),
  sourceCreate: vi.fn(),
  categoryList: vi.fn(),
  categoryCreate: vi.fn(),
}));

vi.mock("@/domain/repositories", () => ({
  iptvChannelRepository: {
    create: mocks.channelCreate,
    delete: mocks.channelDelete,
  },
  iptvSourceRepository: {
    create: mocks.sourceCreate,
  },
  iptvCategoryRepository: {
    list: mocks.categoryList,
    create: mocks.categoryCreate,
  },

vi.mock("@/domain/iptv/m3uDuplicate", async (importOriginal) => {
  const actual = await importOriginal<
    typeof import("@/domain/iptv/m3uDuplicate")
  >();

  return {
    ...actual,
    findM3uDuplicate: vi.fn(() => ({ kind: null })),
  };

enableAutoUnmount(afterEach);

function item(
  name: string,
  url: string,
  groupTitle = "Notícias",
  tvgId = name,
) {
  return {
    name,
    url,
    tvgId,
    tvgName: name,
    tvgLogo: "",
    groupTitle,
    language: "",
    country: "",
    adult: false,
    selected: true,
    errors: [],
  };
}

function mountDialog() {
  return mount(M3uImportDialog, {
    attachTo: document.body,
    props: {
      open: true,
      existing: [],
    },
  });
}

beforeEach(() => {
  Object.values(mocks).forEach((mock) => mock.mockReset());

  mocks.categoryList.mockResolvedValue({
    items: [],
    totalItems: 0,
    totalPages: 1,
    page: 1,
    perPage: 500,
  });

  mocks.categoryCreate.mockImplementation(async (data) => ({
    id: "cat-new",
    ...data,
  }));
});

describe("M3uImportDialog", () => {
  it("não apresenta violações automáticas", async () => {
    const wrapper = mountDialog();

    await flushPromises();
    await expectNoA11yViolations(wrapper.element);
  });

  it("cria categoria a partir de group-title e associa ao canal", async () => {
    mocks.categoryCreate.mockResolvedValue({
      id: "cat1",
      nome: "Notícias",
    });
    mocks.channelCreate.mockResolvedValue({ id: "c1" });
    mocks.sourceCreate.mockResolvedValue({ id: "s1" });

    const wrapper = mountDialog();
    (wrapper.vm as any).items = [
      item("Canal 1", "https://x/1.m3u8"),
    ];

    await (wrapper.vm as any).run();
    await flushPromises();

    expect(mocks.categoryCreate).toHaveBeenCalledWith({
      nome: "Notícias",
      ativo: true,
      ordem: 0,
    });
    expect(mocks.channelCreate).toHaveBeenCalledWith(
      expect.objectContaining({ categoria: "cat1" }),
    );
  });

  it("reutiliza categoria existente ignorando acentos e maiúsculas", async () => {
    mocks.categoryList.mockResolvedValue({
      items: [{ id: "cat1", nome: "Noticias" }],
      totalItems: 1,
      totalPages: 1,
      page: 1,
      perPage: 500,
    });
    mocks.channelCreate.mockResolvedValue({ id: "c1" });
    mocks.sourceCreate.mockResolvedValue({ id: "s1" });

    const wrapper = mountDialog();
    (wrapper.vm as any).items = [
      item("Canal 1", "https://x/1.m3u8", "NOTÍCIAS"),
    ];

    await (wrapper.vm as any).run();

    expect(mocks.categoryCreate).not.toHaveBeenCalled();
    expect(mocks.channelCreate).toHaveBeenCalledWith(
      expect.objectContaining({ categoria: "cat1" }),
    );
  });

  it("ignora duplicados internos e continua o lote", async () => {
    mocks.channelCreate.mockResolvedValue({ id: "c1" });
    mocks.sourceCreate.mockResolvedValue({ id: "s1" });

    const wrapper = mountDialog();
    (wrapper.vm as any).items = [
      item("A", "https://x/a", "Grupo", "a"),
      item("A2", "https://x/a", "Grupo", "b"),
    ];

    await (wrapper.vm as any).run();
    await flushPromises();

    expect(mocks.categoryCreate).toHaveBeenCalledTimes(1);
    expect(mocks.channelCreate).toHaveBeenCalledTimes(1);
    expect(mocks.sourceCreate).toHaveBeenCalledTimes(1);
    expect(wrapper.text()).toContain("1 ignoradas");
    expect(wrapper.text()).toContain("1 duplicados internos");
  });
});
